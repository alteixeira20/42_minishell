# ========================
# Echo Command & Quoting
# ========================
echo Hello World 42
echo "hello    world"
echo 'Hello World'
echo 'Hello World' # shouldn't be printed
echo Hello World # shouldn't be printed
echo "hello "'world'
echo >
echo hi >
echo hi >>
echo hi <
echo hi <<
echo $HOME
echo "$HOME"
echo '$HOME'
echo "$7asdfsasdf"
echo "Hello $USER"
echo '$USER "$HOME"'
echo "$USER $HOME"
echo "user is $USER and home is $HOME"
echo $USER!
echo ""$USER""
echo "hello | world"
echo hello\|world

# ========================
# Redirection with files
# ========================
echo before > tmp_test.txt
ls tmp_test.txt
cat < tmp_test.txt
echo another >> tmp_test.txt
cat < tmp_test.txt
rm tmp_test.txt

# Redirection edge cases
echo first > file > file2
cat < file2
rm file file2

echo one>space.txt
echo two >>space.txt
cat <space.txt
rm space.txt
